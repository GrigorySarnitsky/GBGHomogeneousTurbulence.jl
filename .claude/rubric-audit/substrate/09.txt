"""

# Mask strategies (TwoThirds / HouLi): one evaluation, the low-pass kernel folded into the
# projection's own pass (the kernel-composed ℙ arity) — one pass fewer than project-then-convolve.
function _velocity_nonlinear!(::DealiasingStrategy, nonlinear, generator, û, u, ω, domain::FixedDomain)
  _velocity_lamb!(nonlinear, u, ω, domain, generator.product_spectral, domain.dealias)
  return nonlinear
end

# Phase-shift: average the primary and the half-cell-displaced evaluation (cancels the odd-parity
# aliases); the spherical `domain.dealias` truncation removes the surviving two-axis aliases, folded
# into the passes rather than swept separately — the primary carries it inside its projection, and
# the displaced term picks it up through the composed shift-back∘truncation kernel of the averaging
# convolve, so no whole-field truncation pass remains.
function _velocity_nonlinear!(::PhaseShift, nonlinear, generator, û, u, ω, domain::FixedDomain)
  phase_shift = generator.phase_shift
  product_spectral = generator.product_spectral
  T = real(eltype(field_data(nonlinear)))
  _velocity_lamb!(nonlinear, u, ω, domain, product_spectral, domain.dealias)                          # primary, truncated
  convolve!(phase_shift.velocity_shifted, û, phase_shift.to_shifted)                                   # û on the displaced grid
  to_physical!(phase_shift.u_shifted, phase_shift.velocity_shifted, domain.velocity_plans, product_spectral)
  vorticity_physical!(phase_shift.omega_shifted, phase_shift.velocity_shifted, domain, product_spectral)
  # The displaced Lamb consumes ω_shifted only, so u_shifted still holds this stage's displaced
  # velocity — left fresh for the scalar assembly of the SAME stage (the freshness contract that
  # deletes its re-recovery, decision 45).
  generator.displaced_velocity_fresh = true
  _velocity_lamb!(phase_shift.velocity_shifted, phase_shift.u_shifted, phase_shift.omega_shifted, domain, product_spectral)
  # Shift-back, two-grid average, and truncation in ONE α/β convolve: ½·(𝔻⊙back)⊙(displaced) +
  # ½·(primary, already truncated) — the composed kernel is built once at construction.
  convolve!(nonlinear, phase_shift.velocity_shifted, phase_shift.back_dealiased, T(1 // 2), T(1 // 2))
  return nonlinear
end

function integrate_fixed(integrator, domain, û0, conditions, dt, nsteps)
  buffers = allocate_buffers(integrator, domain; scalar = true)
  generator = ResolvedGenerator(domain, conditions; scalar = true)
  cache = ObservablesCache(domain; scalar = true)
  û = new_vector_field_spectral(domain)
  copyto!(field_data(û), field_data(û0))
  state = ResolvedFields(û, new_scalar_field_spectral(domain))
  deposit!(cache, state, domain, buffers)                 # the seeding deposit
  for _ in 1:nsteps
    step!(integrator, state, generator, domain, buffers, cache, dt)
    deposit!(cache, state, domain, buffers)               # the closing deposit ends the iteration
  end
  return û
end

diffnorm(a, b) = sqrt(sum(abs2, field_data(a) .- field_data(b)))


@testset "integrators" begin
  T = Float64
  N = 16
  L = 2π
  ν = 0.1
  domain = FixedDomain(grid_points_per_axis = N, side_length = L, element_type = T, dealiasing = TwoThirds(), array_type = Array)
  buffers = allocate_buffers(IFLowStorageRK3(), domain; scalar = true)

  û0 = new_vector_field_spectral(domain)
  random_solenoidal_velocity!(û0, domain; kinetic_energy = 0.5, peak_wavenumber = 2.0, rng = MersenneTwister(7))
  dealias!(û0, domain)

  Tfinal = 0.5
  n = 8

  @testset "temporal convergence order" begin
    unforced = FlowConditions(viscosity = ν, schmidt = 1.0, velocity_forcing = NoForcing(), scalar_forcing = NoForcing())
    rk3 = IFLowStorageRK3()
    u1 = integrate_fixed(rk3, domain, û0, unforced, Tfinal / n, n)
    u2 = integrate_fixed(rk3, domain, û0, unforced, Tfinal / (2n), 2n)
    u4 = integrate_fixed(rk3, domain, û0, unforced, Tfinal / (4n), 4n)
    rk3_ratio = diffnorm(u1, u2) / diffnorm(u2, u4)

    rk2 = IFRK2()
    v1 = integrate_fixed(rk2, domain, û0, unforced, Tfinal / n, n)
    v2 = integrate_fixed(rk2, domain, û0, unforced, Tfinal / (2n), 2n)
    v4 = integrate_fixed(rk2, domain, û0, unforced, Tfinal / (4n), 4n)
    rk2_ratio = diffnorm(v1, v2) / diffnorm(v2, v4)

    @test 5.0 < rk3_ratio < 12.0      # third order ⇒ ≈ 8
    @test 3.0 < rk2_ratio < 6.0       # second order ⇒ ≈ 4
    @test rk3_ratio > rk2_ratio       # RK3 converges faster than RK2
  end
